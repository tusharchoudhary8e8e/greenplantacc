━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Tables (all with user_id RLS):
- ma_customers   (uuid PK, org_id, name, phone, email, city, state, pincode, zone, 
                  size_category, crop_types jsonb, gstin, address, opening_balance)
- ma_products    (uuid PK, name, category, unit, variants jsonb [{name,price}], description)
- ma_orders      (uuid PK, order_no, customer_id FK, order_date, status, transport_charge, 
                  advance_payment, foc_amount, paid_amount, items_total, total_amount, 
                  due_amount GENERATED, narration)
- ma_order_items (uuid PK, order_id FK, product_name, variant_name, price, quantity,
                  dispatch_from, dispatch_to, sowing_date, batch_id, dispatched_qty,
                  remaining_qty GENERATED, status)
- ma_batches     (uuid PK, batch_no, product_name, variant_name, sowing_date, 
                  total_seeds, cocopeat_used, trays_used, expected_plants, actual_plants,
                  germination_pct GENERATED, status, notes)
- ma_dispatch    (uuid PK, dispatch_no, order_id FK, customer_id FK, dispatch_date,
                  vehicle_no, driver_name, driver_phone, status, notes)
- ma_dispatch_items (uuid PK, dispatch_id FK, order_item_id FK, product_name, variant_name, quantity)
- ma_payments    (uuid PK, payment_no, customer_id FK, order_id FK, type, amount, 
                  payment_date, method, reference_no, narration)
- ma_employees   (uuid PK, emp_id, name, role, phone, email, department, join_date, salary, status)
- ma_campaigns   (uuid PK, name, type, target_zone, start_date, end_date, budget, status, notes)
- ma_quotes      (uuid PK, quote_no, customer_id FK, customer_name, quote_date, valid_until,
                  status, items jsonb, total_amount, notes)

Performance optimization:
- All tables have indexes on user_id, status, date columns
- Compound indexes: (user_id, status), (user_id, order_date DESC)
- Full-text search index on ma_customers (name, phone, city)
- Row Level Security (RLS) on all tables
- Use .select('id, name, phone, city, zone, size_category') for list queries (not SELECT *)
- Use .range(from, to) for pagination (50 rows per page)
- Debounce search inputs 300ms
- Cache customer/product lists in memory (Map with TTL 5 min)
- Batch inserts: chunk XLSX data into groups of 100

- Smooth transitions between screens (opacity fade 200ms)
- Active sidebar item highlighted
